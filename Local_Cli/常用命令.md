# 常用命令

```bash
# 运行服务端（ACP 协议模式，前端 IDE 以子进程拉起）
uv run local-cli acp

# 运行服务端（PySide6 GUI 桌面模式）
uv run local-cli gui

# 枚举本地 GGUF 模型别名（扫描配置的 model_dir，缺目录/空目录非零退出）
uv run local-cli models

# GGUF 真实链路冒烟（自动写 .temp/test_config.toml，指向本机模型目录；
# 模型目录非 ~/models 时用 LOCAL_CLI_MODEL_DIR 指定；含真实模型工具轮，
# 硬断言收尾/连接，软断言文件落盘——模型不守围栏格式只警告不计挂）
LOCAL_CLI_MODEL_DIR="$HOME/models/qwen3.5-gguf" uv run python tools/smoke_gguf.py

# 无显示服务器环境下验证 GUI 启动
QT_QPA_PLATFORM=offscreen uv run python -m local_cli gui

# 协议回归：spike 断言（应 43 过 / 0 挂；含「协议工具」驱动的真实工具循环节
# 与工具轮消息流无围栏标记的抑制断言）
uv run python tools/spike_handshake.py

# 补充验证：spike 未覆盖项 + parser/executor/config/fence_filter 单元断言（应 31 过 / 0 挂）
uv run python tools/verify_extras.py

# 手工收发协议帧冒烟（stdin 输入 JSON 帧，stdout 只出协议帧，日志在 stderr）
echo '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":1}}' | uv run local-cli acp

# 安装/同步依赖（uv run 会自动同步，一般无需单独执行）
uv sync
```

> 说明：`uv run` 会自动把命令跑在项目 `.venv` 里并同步依赖，等价于手写
> `.venv/bin/...` 路径但更简短。唯一例外：spike 脚本若以 `--bin` 指定被测
> 入口，仍需指向具体可执行文件（如 `--bin .venv/bin/local-cli`）。
